# What the website needs to run, and nothing else.
#
#     pip install -r requirements.txt
#     uvicorn api.main:app --host 0.0.0.0 --port 8000
#
# Needs Python 3.10-3.12; built and tested on 3.12.5. The ceiling is numpy
# 1.26.4, which has no 3.13 wheels -- on 3.13 pip tries to compile it from
# source and fails for want of a Fortran toolchain, which reads as a numpy
# bug rather than a Python version mismatch.
#
# This is the *serving* set, traced from `api.main`: the 26 first-party
# modules reachable from it import these twelve packages and no others. The
# catalog-building side of the repo pulls in pyvo, sparcl, hpgeom and requests
# as well; a host that only serves the site never imports them, so they are
# deliberately absent rather than forgotten.
#
# Pinned exactly, not floated with >=, because the host should run what was
# actually tested. duckdb and pyarrow in particular change query and parquet
# behaviour between minor versions.

# ── the server ───────────────────────────────────────────────────────────────
fastapi==0.141.1
uvicorn==0.51.0
pydantic==2.13.4
orjson==3.12.0
python-dotenv==1.2.3

# ── the filter engine ────────────────────────────────────────────────────────
# DuckDB queries the parquet catalog directly; pyarrow is how it and pandas
# hand rows to each other.
duckdb==1.5.5
pyarrow==21.0.0
pandas==2.2.3
numpy==1.26.4

# ── cutout rendering ─────────────────────────────────────────────────────────
# Cutouts are rendered server-side from FITS into PNG, so the image stack is a
# runtime dependency of the API, not just of the build scripts.
astropy==6.1.3
matplotlib==3.9.2
Pillow==11.0.0

# ── forced photometry ────────────────────────────────────────────────────────
# Adam-Boesky/ztforce, published to PyPI. Installed editable from a local
# checkout on the development machine; the pinned release here is
# byte-identical to tag v0.4.0.
ztforce==0.4.0
